Task Management App

Next.jsTypeScriptPostgreSQLPrisma

Collaborative task manager with real-time updates, drag-and-drop boards, team assignments, and analytics dashboard. Deployed on Vercel.

tasks.ts — API Route 
import { prisma } from '@/lib/prisma';
import { NextResponse } from 'next/server';

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const boardId = searchParams.get('boardId');

  const tasks = await prisma.task.findMany({
    where: { boardId },
    include: { assignee: true, labels: true },
    orderBy: { position: 'asc' }
  });

  return NextResponse.json(tasks);
}

export async function POST(request: Request) {
  const body = await request.json();
  const task = await prisma.task.create({
    data: {
      title: body.title,
      description: body.description,
      boardId: body.boardId,
      assigneeId: body.assigneeId,
      priority: body.priority || 'MEDIUM'
    }
  });
  return NextResponse.json(task, { status: 201 });
}

Weather Dashboard

Real-time weather dashboard with location search, 7-day forecast, and interactive charts. Consumes OpenWeather API and displays data beautifully.

views.py — Django View 
from django.http import JsonResponse
import requests, os

def get_weather(request):
    city = request.GET.get('city', 'Kigali')
    api_key = os.environ.get('WEATHER_API_KEY')
    
    url = f'https://api.openweathermap.org/data/2.5/forecast'
    params = {
        'q': city,
        'appid': api_key,
        'units': 'metric'
    }
    
    response = requests.get(url, params=params)
    data = response.json()
    
    forecast = [{
        'date': item['dt_txt'],
        'temp': item['main']['temp'],
        'description': item['weather'][0]['description'],
        'icon': item['weather'][0]['icon']
    } for item in data['list'][:7]]
    
    return JsonResponse({
        'city': city,
        'forecast': forecast
    })
